CREATE TABLE contacts (
    id SERIAL PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL,
    phone1 VARCHAR(20) NOT NULL,
    middle_name VARCHAR(100),
    birth_date VARCHAR(20),
    address VARCHAR(200),
    phone2 VARCHAR(20),
    phone3 VARCHAR(20)
);
